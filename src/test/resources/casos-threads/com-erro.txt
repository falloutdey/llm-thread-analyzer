import java.util.concurrent.locks.*;
public class Condicao {
    private final Lock lock = new ReentrantLock();
    private final Condition cheio = lock.newCondition();
    private boolean temItem;
    public void tirar() throws InterruptedException {
        lock.lock();
        try { if (!temItem) cheio.await(); temItem = false; } finally { lock.unlock(); }
    }
    public static void main(String[] args) {}
}

### 32_deadlock_transferencia Banco SRC_LOCK_ORDER_DEADLOCK
public class Banco {
    static class Conta { int saldo = 100; }
    static void transferir(Conta de, Conta para, int valor) {
        synchronized (de) {
            synchronized (para) {
                de.saldo -= valor;
                para.saldo += valor;
            }
        }
    }
    public static void main(String[] args) {
        Conta a = new Conta(), b = new Conta();
        new Thread(() -> transferir(a, b, 10)).start();
        new Thread(() -> transferir(b, a, 10)).start();
    }
}

### 34_executor_contador_compartilhado ContadorExecutor SRC_UNSYNCHRONIZED_SHARED_WRITE

import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;
public class ContadorExecutor {
    static int contador;
    public static void main(String[] args) throws InterruptedException {
        ExecutorService ex = Executors.newFixedThreadPool(4);
        for (int i = 0; i < 100; i++) ex.submit(() -> contador++);
