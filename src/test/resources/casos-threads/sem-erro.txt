### n05_produtor_consumidor Caixa -
public class Caixa {
    private Integer item;
    public synchronized void colocar(int x) throws InterruptedException {
        while (item != null) wait();
        item = x;
        notifyAll();
    }
    public synchronized int retirar() throws InterruptedException {
        while (item == null) wait();
        int x = item;
        item = null;
        notifyAll();
        return x;
    }
    public static void main(String[] args) {
        Caixa c = new Caixa();
        new Thread(() -> {
            try { for (int i = 0; i < 3; i++) c.colocar(i); } catch (InterruptedException e) { Thread.currentThread().interrupt(); }
        }).start();
        new Thread(() -> {
            try { for (int i = 0; i < 3; i++) System.out.println(c.retirar()); } catch (InterruptedException e) { Thread.currentThread().interrupt(); }
        }).start();
    }
}
### n06_colecoes_thread_safe ColecoesSeguras -
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.concurrent.CopyOnWriteArrayList;
public class ColecoesSeguras {
    public static void main(String[] args) throws InterruptedException {
        List<Integer> sincronizada = Collections.synchronizedList(new ArrayList<>());
        List<Integer> cow = new CopyOnWriteArrayList<>();
        Runnable r = () -> { for (int i = 0; i < 100; i++) { sincronizada.add(i); cow.add(i); } };
        Thread a = new Thread(r), b = new Thread(r);
        a.start(); b.start(); a.join(); b.join();
        System.out.println(sincronizada.size() + cow.size());
    }
}

### n11_reentrant_lock ContadorComLock -
import java.util.concurrent.locks.Lock;
import java.util.concurrent.locks.ReentrantLock;
public class ContadorComLock {
    private static final Lock lock = new ReentrantLock();
    private static int contador;
    public static void main(String[] args) throws InterruptedException {
        Runnable r = () -> {
            for (int i = 0; i < 1000; i++) {
                lock.lock();
                try { contador++; } finally { lock.unlock(); }
            }
        };
        Thread a = new Thread(r), b = new Thread(r);
        a.start(); b.start(); a.join(); b.join();
        System.out.println(contador);
    }
}

### n16_transferencia_com_ordem_fixa BancoOrdenado -
public class BancoOrdenado {
    static class Conta { final int id; int saldo = 100; Conta(int id) { this.id = id; } }
    static void transferir(Conta de, Conta para, int valor) {
        Conta primeiro = de.id < para.id ? de : para;
        Conta segundo = de.id < para.id ? para : de;
        synchronized (primeiro) {
            synchronized (segundo) {
                de.saldo -= valor;
                para.saldo += valor;
            }
        }
    }
    public static void main(String[] args) {
        Conta a = new Conta(1), b = new Conta(2);
        new Thread(() -> transferir(a, b, 10)).start();
        new Thread(() -> transferir(b, a, 10)).start();
    }
}
